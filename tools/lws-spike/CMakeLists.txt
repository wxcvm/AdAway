# Isolated libwebsockets feasibility harness. Not linked into AdAway.
cmake_minimum_required(VERSION 3.22)
project(adaway_lws_spike C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(ANDROID)
  set(LWS_PLAT_ANDROID ON CACHE BOOL "" FORCE)
  # The pinned config template omits this C macro although its CMake source
  # list uses it. Supply the matching public platform flag, no source patch.
  add_compile_definitions(LWS_PLAT_ANDROID=1)
endif()
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL REQUIRED)
option(SPIKE_PREFLIGHT "Allow a non-baseline OpenSSL for local compile diagnostics only" OFF)
if(NOT SPIKE_PREFLIGHT AND NOT OPENSSL_VERSION STREQUAL "3.5.9")
  message(FATAL_ERROR "Measured builds require pinned OpenSSL 3.5.9, got ${OPENSSL_VERSION}")
endif()
if(NOT IS_DIRECTORY "${LWS_SOURCE}" OR NOT IS_DIRECTORY "${MONGOOSE_SOURCE}")
  message(FATAL_ERROR "Use fetch.py and supply LWS_SOURCE and MONGOOSE_SOURCE")
endif()
# MinGW's detected pthread types require the standard header before LWS headers.
# This changes compilation flags only, not upstream source or callback internals.
if(MINGW)
  add_compile_options(-include pthread.h -pthread)
endif()
if(WIN32)
  # Winsock defaults to 64 fd_set entries, below this fixture's common
  # 128 inbound + 64 controlled-forward allowance. Set before any headers.
  add_compile_definitions(FD_SETSIZE=256)
endif()
set(LWS_MAX_SMP 1 CACHE STRING "Single service thread matches current production model" FORCE)
set(LWS_WITH_STATIC ON CACHE BOOL "" FORCE)
set(LWS_WITH_SHARED OFF CACHE BOOL "" FORCE)
set(LWS_WITH_SSL ON CACHE BOOL "" FORCE)
set(LWS_WITH_HTTP_PROXY ON CACHE BOOL "Controlled loopback proxy probe" FORCE)
set(LWS_CLIENT_HTTP_PROXYING OFF CACHE BOOL "No environment-selected external proxy in local fixture" FORCE)
set(LWS_WITH_GENCRYPTO ON CACHE BOOL "" FORCE)
set(LWS_WITH_SCHANNEL OFF CACHE BOOL "" FORCE)
set(LWS_WITH_HTTP3 OFF CACHE BOOL "" FORCE)
set(LWS_ROLE_QUIC OFF CACHE BOOL "" FORCE)
set(LWS_WITH_HTTP2 OFF CACHE BOOL "" FORCE)
set(LWS_WITHOUT_TESTAPPS ON CACHE BOOL "" FORCE)
set(LWS_WITH_MINIMAL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LWS_WITHOUT_EXTENSIONS ON CACHE BOOL "" FORCE)
foreach(feature LHP STUB SPAWN CGI SECURE_STREAMS SYS_STATE SYS_SMD UPNG JPEG DLO GZINFLATE ZLIB BUNDLED_ZLIB PLUGINS EVLIB_PLUGINS LIBUV LIBEVENT LIBEV HTTP_BASIC_AUTH HTTP_DIGEST_AUTH)
  set(LWS_WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(DISABLE_WERROR ON CACHE BOOL "Upstream warnings retained in logs; spike adapters use Werror" FORCE)
option(SPIKE_VERBOSE_BUILD "Extra diagnostic logs, excluded from measured builds" OFF)
if(SPIKE_VERBOSE_BUILD)
  set(LWS_LOGGING_BITFIELD_SET 264 CACHE STRING "Diagnostic INFO and CLIENT" FORCE)
else()
  set(LWS_LOGGING_BITFIELD_SET 0 CACHE STRING "Measured release logging" FORCE)
endif()
add_subdirectory("${LWS_SOURCE}" lws-build EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
set(common src/main.c src/fixture.c src/certificates.c)
add_executable(spike_probe src/probe.c)
target_link_libraries(spike_probe PRIVATE OpenSSL::SSL OpenSSL::Crypto)
add_executable(spike_lws ${common} src/backend_lws.c)
add_executable(spike_mongoose ${common} src/backend_mongoose.c "${MONGOOSE_SOURCE}/mongoose.c")
if(NOT MSVC)
  # Keep vendor diagnostics, just as with DISABLE_WERROR for LWS. Only our
  # adapters/probe are warnings-as-errors; do not patch the baseline library.
  set_source_files_properties("${MONGOOSE_SOURCE}/mongoose.c" PROPERTIES COMPILE_OPTIONS "-Wno-error")
endif()
target_include_directories(spike_lws PRIVATE src)
target_include_directories(spike_mongoose PRIVATE src "${MONGOOSE_SOURCE}")
target_compile_definitions(spike_mongoose PRIVATE MG_ENABLE_IPV6=1 MG_IPV6_V6ONLY=1 MG_TLS=MG_TLS_OPENSSL MG_MAX_RECV_SIZE=16384)
if(WIN32)
  # Match the existing Windows production build's public event backend.
  target_compile_definitions(spike_mongoose PRIVATE MG_ENABLE_POLL=1)
endif()
target_link_libraries(spike_lws PRIVATE websockets OpenSSL::SSL OpenSSL::Crypto)
target_link_libraries(spike_mongoose PRIVATE OpenSSL::SSL OpenSSL::Crypto)
foreach(t spike_lws spike_mongoose spike_probe)
  if(NOT MSVC)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
  endif()
  if(WIN32)
    target_link_libraries(${t} PRIVATE ws2_32 crypt32 bcrypt iphlpapi winmm pthread)
  else()
    target_compile_definitions(${t} PRIVATE _GNU_SOURCE)
    target_link_libraries(${t} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  endif()
endforeach()
